GENERATORS = ("oscillator", "lorentz", "relativistic", "kepler", "blackbody")
BASE_NOISE_LEVELS = (0.01, 0.05, 0.10, 0.20, 0.40, 0.80)
SELECTION_POLICIES = ("mdl_diag", "cv_rmse_then_dl", "cv_rank_stable")

@dataclass(frozen=True)
class BankEntry:
    spec: CandidateSpec
    template_family: str
    truth_for: str
    distractor_class: str


def _cube(expr: Expr) -> Expr:
    return op2("mul", op2("mul", expr, expr), expr)


def _square(expr: Expr) -> Expr:
    return op2("mul", expr, expr)


def _clip(expr: Expr, low: float, high: float) -> Expr:
    return op3("clip", expr, Const(low), Const(high))


def _build_bank(bank_size: int, seed: int = 20260822) -> tuple[list[BankEntry], dict[str, str]]:
    variables = ("x1", "x2", "x3")
    staged: dict[str, tuple[list[Expr], str, str, str]] = {}

    def add(basis: list[Expr], family: str, truth_for: str = "", distractor: str = "near_equivalent") -> None:
        signature = json.dumps([expr.to_string() for expr in basis], separators=(",", ":"))
        if signature not in staged or truth_for:
            staged[signature] = (basis, family, truth_for, distractor)

    for first, second, third in permutations(variables, 3):
        x1, x2, x3 = Var(first), Var(second), Var(third)
        for frequency in (0.7, 0.9, 1.0, 1.1, 1.3, 1.5, 1.8):
            for second_op in ("cos", "sin"):
                basis = [
                    Const(1.0),
                    op1("sin", op2("mul", Const(frequency), x1)),
                    op1(second_op, x2),
                    x3,
                ]
                exact = first == "x1" and second == "x2" and third == "x3" and frequency == 1.1 and second_op == "cos"
                add(basis, "oscillator", "oscillator" if exact else "", "frequency_or_operator")

        for denominator_power in (1, 2, 3, 4):
            for denominator_offset in (0.5, 1.0, 2.0):
                denominator_term = x3
                for _ in range(denominator_power - 1):
                    denominator_term = op2("mul", denominator_term, x3)
                basis = [
                    Const(1.0),
                    op2(
                        "div",
                        op2("mul", x1, x2),
                        op2("add", Const(denominator_offset), denominator_term),
                    ),
                ]
                exact = first == "x1" and second == "x2" and third == "x3" and denominator_power == 2 and denominator_offset == 1.0
                add(basis, "lorentz", "lorentz" if exact else "", "variable_power_or_offset")

        for beta in (0.06, 0.08, 0.09765625, 0.12, 0.16):
            for exponent in (-1.0, -0.5):
                relativistic = op2(
                    "mul",
                    x1,
                    op2(
                        "pow",
                        _clip(
                            op2("sub", Const(1.0), op2("mul", Const(beta), _square(x2))),
                            0.06,
                            4.0,
                        ),
                        Const(exponent),
                    ),
                )
                basis = [Const(1.0), relativistic, x3]
                exact = first == "x1" and second == "x2" and third == "x3" and beta == 0.09765625 and exponent == -0.5
                add(basis, "relativistic", "relativistic" if exact else "", "coefficient_exponent_or_variable")

    for first, second in permutations(variables, 2):
        numerator, denominator = Var(first), Var(second)
        for denominator_power in (2, 3, 4):
            denominator_term = denominator
            for _ in range(denominator_power - 1):
                denominator_term = op2("mul", denominator_term, denominator)
            for exponent in (1.0 / 3.0, 0.5, 2.0 / 3.0):
                basis = [
                    Const(1.0),
                    op2(
                        "pow",
                        op2(
                            "div",
                            _clip(numerator, 0.1, 10.0),
                            _clip(denominator_term, 0.05, 100.0),
                        ),
                        Const(exponent),
                    ),
                ]
                exact = first == "x1" and second == "x2" and denominator_power == 3 and exponent == 0.5
                add(basis, "kepler", "kepler" if exact else "", "power_exponent_or_variable")

    for first, second, third in permutations(variables, 3):
        numerator, exponent_var, additive = Var(first), Var(second), Var(third)
        for numerator_power in (2, 3, 4):
            numerator_term = numerator
            for _ in range(numerator_power - 1):
                numerator_term = op2("mul", numerator_term, numerator)
            for exponent_scale in (0.8, 1.0, 1.2):
                denominator = op2(
                    "add",
                    op2(
                        "sub",
                        op1("exp", _clip(op2("mul", Const(exponent_scale), exponent_var), 0.0, 5.0)),
                        Const(1.0),
                    ),
                    Const(1.0e-2),
                )
                basis = [Const(1.0), op2("div", numerator_term, denominator), additive]
                exact = first == "x1" and second == "x2" and third == "x3" and numerator_power == 3 and exponent_scale == 1.0
                add(basis, "blackbody", "blackbody" if exact else "", "power_scale_or_variable")

    exact_rows = [(signature, payload) for signature, payload in staged.items() if payload[2]]
    distractors = [(signature, payload) for signature, payload in staged.items() if not payload[2]]
    if len(exact_rows) != len(GENERATORS):
        raise AssertionError(f"expected five truth templates, found {len(exact_rows)}")
    if bank_size < 200 or bank_size > len(staged):
        raise ValueError(f"bank_size must be in [200, {len(staged)}]")
    rng = np.random.default_rng(seed)
    chosen_idx = rng.choice(len(distractors), size=bank_size - len(exact_rows), replace=False)
    selected = exact_rows + [distractors[int(index)] for index in chosen_idx]
    rng.shuffle(selected)

    entries: list[BankEntry] = []
    truth_ids: dict[str, str] = {}
    for offset, (_, payload) in enumerate(selected, start=401):
        basis, family, truth_for, distractor = payload
        candidate_id = f"cand_{offset:04d}"
        entry = BankEntry(
            spec=CandidateSpec(
                candidate_id=candidate_id,
                basis_exprs=basis,
                sigma_mode="constant",
                ridge=1.0e-8,
            ),
            template_family=family,
            truth_for=truth_for,
            distractor_class="truth" if truth_for else distractor,
        )
        entries.append(entry)
        if truth_for:
            truth_ids[truth_for] = candidate_id
    return entries, truth_ids


def _generate(generator: str, sigma: float, seed: int, n_per_env: int = 120) -> pd.DataFrame:
    rng = np.random.default_rng(seed)
    rows: list[pd.DataFrame] = []
    for env in range(5):
        x1 = rng.uniform(0.2 + 0.05 * env, 2.5 + 0.3 * env, size=n_per_env)
        x2 = rng.uniform(0.1, 2.0 + 0.2 * env, size=n_per_env)
        x3 = rng.normal(0.0, 1.0 + 0.15 * env, size=n_per_env)
        if generator == "oscillator":
            truth = 2.2 * np.sin(1.1 * x1) + 0.8 * np.cos(x2) - 0.2 * x3
        elif generator == "lorentz":
            truth = (x1 * x2) / (1.0 + x3**2)
        elif generator == "relativistic":
            truth = x1 / np.sqrt(np.maximum(1.0 - (x2 / 3.2) ** 2, 0.06)) + 0.1 * x3
        elif generator == "kepler":
            truth = np.sqrt(np.maximum(x1, 0.1) / np.maximum(x2**3, 0.05))
        elif generator == "blackbody":
            truth = (x1**3) / (np.exp(np.clip(x2, 0.0, 5.0)) - 1.0 + 1.0e-2) + 0.05 * x3
        else:
            raise ValueError(f"unknown generator: {generator}")
        y = truth + rng.normal(0.0, sigma, size=n_per_env)
        rows.append(
            pd.DataFrame(
                {
                    "env_id": env,
                    "x1": x1,
                    "x2": x2,
                    "x3": x3,
                    "truth": truth,
                    "y_tilde": np.round(y / 1.0e-3) * 1.0e-3,
                }
            )
        )
    frame = pd.concat(rows, ignore_index=True)
    frame["split"] = five_role_split_by_env(frame["env_id"].to_numpy(), seed=seed)
    return frame


def _symbols(expressions: list[Expr]) -> tuple[set[str], set[str]]:
    variables: set[str] = set()
    operators: set[str] = set()

    def visit(expr: Expr) -> None:
        if expr.op == "var" and expr.name is not None:
            variables.add(str(expr.name))
        elif expr.op not in {"const", "bool", "var"}:
            operators.add(expr.op)
        for child in expr.args:
            visit(child)

    for expression in expressions:
        visit(expression)
    return variables, operators


def _rmse(y: np.ndarray, prediction: np.ndarray) -> float:
    return float(np.sqrt(np.mean((np.asarray(y, dtype=float) - np.asarray(prediction, dtype=float)) ** 2)))


def _pooled_out_of_sample_rmse(
    discovery_cv_rmse: float,
    diagnosis_rmse: float,
    n_discovery: int,
    n_diagnosis: int,
) -> float:
    total = int(n_discovery) + int(n_diagnosis)
    if total <= 0:
        raise ValueError("pooled out-of-sample RMSE requires at least one observation")
    weighted_mse = (
        int(n_discovery) * float(discovery_cv_rmse) ** 2
        + int(n_diagnosis) * float(diagnosis_rmse) ** 2
    ) / total
    return float(np.sqrt(weighted_mse))


def _diagnosis_invariance(
    model: LinearBasisModel,
    diag: pd.DataFrame,
    features: list[str],
    seed: int,
) -> dict[str, float | bool]:
    mu, sigma = model.predict_mu_sigma(diag[features].to_numpy(float))
    residual = (diag["y_tilde"].to_numpy(float) - mu) / np.maximum(sigma, 1.0e-8)
    env = diag["env_id"].to_numpy(int)
    residual = normalize_residuals_by_env(
        residual,
        env,
        mode=INVARIANCE_RESIDUAL_NORMALIZATION,
        min_scale=1.0e-6,
        shrinkage=4.0,
    )
    result = mmd_rff_permutation_test(
        z=residual,
        env=env,
        n_features=INVARIANCE_RFF_FEATURES,
        n_perm=INVARIANCE_PERMUTATIONS,
        seed=seed,
    )
    return {
        "inv_stat_diag": float(result.stat),
        "p_inv_diag": float(result.p_value),
        "gate_pass": bool(result.p_value >= INVARIANCE_ALPHA),
    }


def _record_sha256(payload: dict[str, Any]) -> str:
    canonical = json.dumps(payload, sort_keys=True, separators=(",", ":"), ensure_ascii=True)
    return hashlib.sha256(canonical.encode("utf-8")).hexdigest()


def _select_and_refit(
    frame: pd.DataFrame,
    bank: list[BankEntry],
    seed: int,
    selection_policy: str,
    crossfit_k: int,
    crossfit_repeats: int,
    trace: dict[str, Any] | None = None,
) -> tuple[BankEntry, LinearBasisModel, dict[str, Any]]:
    """Run the frozen Table 5 selection, gate, ridge, and refit protocol.

    Candidate structures are fitted and cross-fitted on discovery only.  The
    diagnosis role supplies the ordered holdout-RMSE tie-break and the residual
    invariance gate.  After structure selection, only the winner is evaluated
    on the fixed ridge multiplier grid; the tuned structure is finally refitted
    on discovery plus diagnosis.  Audit, calibration, and test are never read.
    """

    if selection_policy not in SELECTION_POLICIES:
        raise ValueError(f"unknown selection policy: {selection_policy}")
    if crossfit_k < 2 or crossfit_repeats < 1:
        raise ValueError("crossfit_k must be at least two and crossfit_repeats positive")

    features = ["x1", "x2", "x3"]
    disc = frame[frame["split"] == "disc"].copy()
    diag = frame[frame["split"] == "diag"].copy()
    allowed = frame[frame["split"].isin(["disc", "diag"])].copy()
    if disc.empty or diag.empty or allowed.shape[0] != disc.shape[0] + diag.shape[0]:
        raise ValueError("recovery selection requires disjoint non-empty disc and diag roles")

    rows: list[dict[str, Any]] = []
    for index, entry in enumerate(bank):
        holdout = _fit_score_single(
            spec=entry.spec,
            df_fit=disc,
            df_eval=diag,
            feature_cols=features,
            target_col="y_tilde",
            delta_y=1.0e-3,
            use_bin_likelihood=True,
            use_fisher=False,
            fisher_probe_points=16,
            fisher_jitter=1.0e-4,
            robust_eps=1.0e-4,
            device="cpu",
            max_fisher_params=16,
            fisher_seed=seed + index,
            strict_param_code=True,
            fixed_param_step=1.0e-3,
        )
        cv_rmse_values: list[float] = []
        if selection_policy != "mdl_diag":
            for repeat in range(crossfit_repeats):
                cf = crossfit_residuals(
                    df=disc,
                    feature_cols=features,
                    target_col="y_tilde",
                    env_col="env_id",
                    model_factory=lambda spec=entry.spec: LinearBasisModel(
                        spec=spec,
                        feature_cols=features,
                        ridge=spec.ridge,
                        device="cpu",
                    ),
                    k=crossfit_k,
                    seed=seed + 1009 * repeat,
                )
                cv_rmse_values.append(_rmse(disc["y_tilde"].to_numpy(float), cf.mu))
        gate = _diagnosis_invariance(
            model=holdout["model"],
            diag=diag,
            features=features,
            seed=seed,
        )
        rows.append(
            {
                "entry": entry,
                "candidate_id": entry.spec.candidate_id,
                "dl_total": float(holdout["dl_total"]),
                "diag_rmse": float(holdout["rmse_eval"]),
                "cv_rmse": float(np.mean(cv_rmse_values)) if cv_rmse_values else float("nan"),
                "cv_sd": float(np.std(cv_rmse_values, ddof=1)) if len(cv_rmse_values) > 1 else 0.0,
                **gate,
            }
        )

    scored = pd.DataFrame(rows)
    valid = scored[scored["gate_pass"]].copy()
    if valid.empty:
        raise RuntimeError(
            "all Table 5 candidates failed the frozen diagnosis residual-invariance gate"
        )
    if selection_policy == "mdl_diag":
        best_index = int(valid.sort_values(["dl_total", "diag_rmse", "candidate_id"]).index[0])
    elif selection_policy == "cv_rmse_then_dl":
        best_index = int(
            valid.sort_values(["cv_rmse", "diag_rmse", "dl_total", "candidate_id"]).index[0]
        )
    else:
        valid["cv_rank"] = valid["cv_rmse"].rank(method="average")
        valid["diag_rank"] = valid["diag_rmse"].rank(method="average")
        valid["dl_rank"] = valid["dl_total"].rank(method="average")
        valid["stable_rank"] = 0.65 * valid["cv_rank"] + 0.25 * valid["diag_rank"] + 0.10 * valid["dl_rank"]
        best_index = int(
            valid.sort_values(
                ["stable_rank", "cv_rmse", "diag_rmse", "dl_total", "candidate_id"]
            ).index[0]
        )
    best_row = scored.loc[best_index]
    best_entry = best_row["entry"]

    ridge_groups: dict[float, list[float]] = {}
    for multiplier in RIDGE_MULTIPLIERS:
        requested = float(best_entry.spec.ridge) * float(multiplier)
        ridge_groups.setdefault(max(RIDGE_FLOOR, requested), []).append(float(multiplier))

    ridge_rows: list[dict[str, Any]] = []
    for effective_ridge, multiplier_group in sorted(ridge_groups.items()):
        multiplier = 1.0 if 1.0 in multiplier_group else multiplier_group[0]
        requested_ridge = float(best_entry.spec.ridge) * float(multiplier)
        variant = replace(best_entry.spec, ridge=effective_ridge)
        holdout = _fit_score_single(
            spec=variant,
            df_fit=disc,
            df_eval=diag,
            feature_cols=features,
            target_col="y_tilde",
            delta_y=1.0e-3,
            use_bin_likelihood=True,
            use_fisher=False,
            fisher_probe_points=16,
            fisher_jitter=1.0e-4,
            robust_eps=1.0e-4,
            device="cpu",
            max_fisher_params=16,
            fisher_seed=seed + len(bank) + int(round(1000.0 * multiplier)),
            strict_param_code=True,
            fixed_param_step=1.0e-3,
        )
        cv_values: list[float] = []
        for repeat in range(crossfit_repeats):
            cf = crossfit_residuals(
                df=disc,
                feature_cols=features,
                target_col="y_tilde",
                env_col="env_id",
                model_factory=lambda spec=variant: LinearBasisModel(
                    spec=spec,
                    feature_cols=features,
                    ridge=spec.ridge,
                    device="cpu",
                ),
                k=crossfit_k,
                seed=seed + 1009 * repeat,
            )
            cv_values.append(_rmse(disc["y_tilde"].to_numpy(float), cf.mu))
        cv_rmse = float(np.mean(cv_values))
        diag_rmse = float(holdout["rmse_eval"])
        gate = _diagnosis_invariance(
            model=holdout["model"],
            diag=diag,
            features=features,
            seed=seed,
        )
        ridge_rows.append(
            {
                "ridge_multiplier": float(multiplier),
                "ridge_multiplier_group": tuple(multiplier_group),
                "requested_ridge": requested_ridge,
                "effective_ridge": effective_ridge,
                "ridge_floor_applied": bool(not np.isclose(effective_ridge, requested_ridge)),
                "ridge_floor_collapsed_group": bool(len(multiplier_group) > 1),
                "cv_rmse": cv_rmse,
                "diag_rmse": diag_rmse,
                "pooled_oos_rmse": _pooled_out_of_sample_rmse(
                    cv_rmse,
                    diag_rmse,
                    n_discovery=disc.shape[0],
                    n_diagnosis=diag.shape[0],
                ),
                "dl_total": float(holdout["dl_total"]),
                "model": holdout["model"],
                **gate,
            }
        )
    ridge_table = pd.DataFrame(ridge_rows)
    ridge_valid = ridge_table[ridge_table["gate_pass"]].copy()
    if ridge_valid.empty:
        raise RuntimeError(
            "all Table 5 ridge variants failed the frozen diagnosis residual-invariance gate"
        )
    tuned = ridge_valid.sort_values(
        ["pooled_oos_rmse", "cv_rmse", "diag_rmse", "dl_total", "effective_ridge"]
    ).iloc[0]

    refit = _fit_score_single(
        spec=tuned["model"].spec,
        df_fit=allowed,
        df_eval=allowed,
        feature_cols=features,
        target_col="y_tilde",
        delta_y=1.0e-3,
        use_bin_likelihood=True,
        use_fisher=False,
        fisher_probe_points=16,
        fisher_jitter=1.0e-4,
        robust_eps=1.0e-4,
        device="cpu",
        max_fisher_params=16,
        fisher_seed=seed + len(bank),
        strict_param_code=True,
        fixed_param_step=1.0e-3,
    )
    diagnostics = {
        "selected_dl_total": float(best_row["dl_total"]),
        "selected_diag_rmse": float(best_row["diag_rmse"]),
        "selected_cv_rmse": float(best_row["cv_rmse"]),
        "selected_cv_sd": float(best_row["cv_sd"]),
        "candidate_gate_evaluated": True,
        "candidate_count": int(scored.shape[0]),
        "candidate_gate_pass_count": int(valid.shape[0]),
        "n_discovery": int(disc.shape[0]),
        "n_diagnosis": int(diag.shape[0]),
        "selected_inv_stat_diag": float(best_row["inv_stat_diag"]),
        "selected_p_inv_diag": float(best_row["p_inv_diag"]),
        "selected_gate_pass": bool(best_row["gate_pass"]),
        "invariance_alpha": INVARIANCE_ALPHA,
        "invariance_rff_features": INVARIANCE_RFF_FEATURES,
        "invariance_permutations": INVARIANCE_PERMUTATIONS,
        "ridge_base": float(best_entry.spec.ridge),
        "postselect_ridge_multiplier": float(tuned["ridge_multiplier"]),
        "postselect_ridge_multiplier_group": ",".join(
            f"{value:g}" for value in tuned["ridge_multiplier_group"]
        ),
        "postselect_requested_ridge": float(tuned["requested_ridge"]),
        "postselect_effective_ridge": float(tuned["effective_ridge"]),
        "postselect_ridge_floor_applied": bool(tuned["ridge_floor_applied"]),
        "postselect_ridge_floor_collapsed_group": bool(tuned["ridge_floor_collapsed_group"]),
        "postselect_cv_rmse": float(tuned["cv_rmse"]),
        "postselect_diag_rmse": float(tuned["diag_rmse"]),
        "postselect_pooled_oos_rmse": float(tuned["pooled_oos_rmse"]),
        "postselect_inv_stat_diag": float(tuned["inv_stat_diag"]),
        "postselect_p_inv_diag": float(tuned["p_inv_diag"]),
        "postselect_gate_pass": bool(tuned["gate_pass"]),
        "ridge_grid_nominal_count": len(RIDGE_MULTIPLIERS),
        "ridge_grid_effective_count": int(ridge_table["effective_ridge"].nunique()),
        "ridge_grid_evaluated_count": int(ridge_table.shape[0]),
    }
    if trace is not None:
        candidate_trace = scored.drop(columns=["entry"]).to_dict(orient="records")
        ridge_trace = ridge_table.drop(columns=["model"]).to_dict(orient="records")
        selection_record = {
            "bank_name": BANK_NAME,
            "candidate_id": best_entry.spec.candidate_id,
            "seed": int(seed),
            "selection_policy": TABLE5_SELECTION_POLICY,
            "selection_order": ["cv_rmse", "diag_rmse", "dl_total", "candidate_id"],
            "structure_fit_roles": ["disc"],
            "crossfit_roles": ["disc"],
            "diagnosis_roles": ["diag"],
            "n_discovery": int(disc.shape[0]),
            "n_diagnosis": int(diag.shape[0]),
            "crossfit_folds": int(crossfit_k),
            "cv_rmse": diagnostics["selected_cv_rmse"],
            "diag_rmse": diagnostics["selected_diag_rmse"],
            "dl_total": diagnostics["selected_dl_total"],
            "gate_p_value": diagnostics["selected_p_inv_diag"],
            "gate_alpha": INVARIANCE_ALPHA,
        }
        refit_record = {
            "bank_name": BANK_NAME,
            "candidate_id": best_entry.spec.candidate_id,
            "seed": int(seed),
            "ridge_multiplier": diagnostics["postselect_ridge_multiplier"],
            "ridge_multiplier_group": [
                float(value) for value in tuned["ridge_multiplier_group"]
            ],
            "effective_ridge": diagnostics["postselect_effective_ridge"],
            "final_refit_roles": ["disc", "diag"],
            "parameter_vector": [float(value) for value in refit["model"].param_vector()],
            "expression": refit["model"].to_expr().to_string(),
        }
        trace.update(
            {
                "candidate_ledger": candidate_trace,
                "ridge_ledger": ridge_trace,
                "selection_record": selection_record,
                "selection_record_sha256": _record_sha256(selection_record),
                "refit_record": refit_record,
                "refit_record_sha256": _record_sha256(refit_record),
            }
        )
    return best_entry, refit["model"], diagnostics


def _run_cell(
    generator: str,
    sigma: float,
    seed: int,
    bank: list[BankEntry],
    truth_ids: dict[str, str],
    selection_policy: str = "mdl_diag",
    crossfit_k: int = 3,
    crossfit_repeats: int = 1,
) -> dict[str, Any]:
    start = time.perf_counter()
    frame = _generate(generator, sigma, seed)
    best_entry, best_model, selection = _select_and_refit(
        frame=frame,
        bank=bank,
        seed=seed,
        selection_policy=selection_policy,
        crossfit_k=crossfit_k,
        crossfit_repeats=crossfit_repeats,
    )
    test = frame[frame["split"] == "test"]
    prediction, _ = best_model.predict_mu_sigma(test[["x1", "x2", "x3"]].to_numpy(float))
    truth = test["truth"].to_numpy(float)
    noisy = test["y_tilde"].to_numpy(float)
    scale = max(float(np.std(truth)), 1.0e-12)
    semantic_nrmse = float(np.sqrt(np.mean((prediction - truth) ** 2)) / scale)
    semantic_max = float(np.max(np.abs(prediction - truth)) / scale)
    selected_vars, selected_ops = _symbols(best_entry.spec.basis_exprs)
    truth_entry = next(entry for entry in bank if entry.spec.candidate_id == truth_ids[generator])
    truth_vars, truth_ops = _symbols(truth_entry.spec.basis_exprs)
    return {
        "generator": generator,
        "noise_sigma": sigma,
        "seed": seed,
        "bank_name": BANK_NAME,
        "bank_size": len(bank),
        "n_distractors": len(bank) - 1,
        "selected_candidate_id": best_entry.spec.candidate_id,
        "selection_policy": (
            TABLE5_SELECTION_POLICY if selection_policy == "cv_rmse_then_dl" else selection_policy
        ),
        "structure_fit_roles": "disc",
        "crossfit_roles": "disc",
        "diagnosis_roles": "diag",
        "final_refit_roles": "disc+diag",
        **selection,
        "recovered": best_entry.spec.candidate_id == truth_ids[generator],
        "semantic_nrmse": semantic_nrmse,
        "semantic_max_scaled_error": semantic_max,
        "variable_precision": len(selected_vars & truth_vars) / max(len(selected_vars), 1),
        "variable_recall": len(selected_vars & truth_vars) / max(len(truth_vars), 1),
        "operator_precision": len(selected_ops & truth_ops) / max(len(selected_ops), 1),
        "operator_recall": len(selected_ops & truth_ops) / max(len(truth_ops), 1),
        "rmse_test": float(np.sqrt(np.mean((prediction - noisy) ** 2))),
        "runtime_sec": time.perf_counter() - start,
    }


def _write_mapping(bank: list[BankEntry], out: Path) -> None:
    rows = []
    for entry in bank:
        variables, operators = _symbols(entry.spec.basis_exprs)
        rows.append(
            {
                "bank_name": BANK_NAME,
                "candidate_id": entry.spec.candidate_id,
                "template_family": entry.template_family,
                "truth_for": entry.truth_for,
                "distractor_class": entry.distractor_class,
                "basis_preorder": json.dumps(
                    [expr.to_string() for expr in entry.spec.basis_exprs],
                    separators=(",", ":"),
                ),
                "variables": ",".join(sorted(variables)),
                "operators": ",".join(sorted(operators)),
            }
        )
    pd.DataFrame(rows).to_csv(out, index=False)
